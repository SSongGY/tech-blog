===== 3. 필드 목록 — 범위, 순서, 여집합
$ cut -d, -f1,3 users.csv
# 사용자 목록 — 이 줄에는 쉼표가 없다
id,dept
1,dev
2,ops
3,dev
$ cut -d, -f3,1 users.csv
# 사용자 목록 — 이 줄에는 쉼표가 없다
id,dept
1,dev
2,ops
3,dev
$ cut -d, -f2- users.csv
# 사용자 목록 — 이 줄에는 쉼표가 없다
name,dept,email
kim,dev,kim@example.com
lee,ops,lee@example.com
park,dev,park@example.com
$ cut -d, -f-2 users.csv
# 사용자 목록 — 이 줄에는 쉼표가 없다
id,name
1,kim
2,lee
3,park
$ cut -d, -f2 --complement users.csv
# 사용자 목록 — 이 줄에는 쉼표가 없다
id,dept,email
1,dev,kim@example.com
2,ops,lee@example.com
3,dev,park@example.com
$ cut -d, -f1,2 --output-delimiter=' | ' users.csv
# 사용자 목록 — 이 줄에는 쉼표가 없다
id | name
1 | kim
2 | lee
3 | park

===== 4. 구분자가 없는 줄 — 기본은 그대로, -s 는 버린다

$ cut -d, -f2 -s users.csv | head -2

===== 7. -b 와 -c — 이 판에서는 둘 다 바이트다
$ LC_ALL=en_US.UTF-8 cut -c1-3 hangul.txt | od -An -tx1
 ed 95 9c 0a
$ LC_ALL=en_US.UTF-8 cut -b1-3 hangul.txt | od -An -tx1
 ed 95 9c 0a
$ LC_ALL=en_US.UTF-8 cut -c1-2 hangul.txt | od -An -tx1
 ed 95 0a
$ cut -b1-3 --complement hangul.txt | od -An -tx1
 ea b8 80 eb a1 9c ea b7 b8 0a
$ cut -c1-3 users.csv | tail -3

===== 8. -z — NUL 로 끝나는 레코드

===== 9. paste — 줄을 옆으로 붙인다
$ paste names.txt pays.txt
kim	700
lee	450
park	500
$ paste -d, names.txt depts.txt pays.txt
kim,dev,700
lee,ops,450
park,,500
$ paste -d',;' names.txt depts.txt pays.txt
kim,dev;700
lee,ops;450
park,;500
$ paste -s names.txt
kim	lee	park
$ paste -sd, names.txt
kim,lee,park
$ paste -sd+ pays.txt
700+450+500
$ paste - - < letters.txt
a	b
c	d
e	f
$ paste -d, - - - < letters.txt
a,b,c
d,e,f
$ cut -d, -f2 -s users.csv | paste -sd' '
name kim lee park
$ printf 'a\0b\0' | paste -z -sd, | od -An -c
   a   ,   b  \0

===== 10. tr — 글자를 바꾼다
$ echo 'Hello World' | tr a-z A-Z
HELLO WORLD
$ echo 'Hello World' | tr '[:lower:]' '[:upper:]'
HELLO WORLD
$ echo 'abcabc' | tr abc x
xxxxxx
$ echo 'abcabc' | tr -t abc x
xbcxbc
$ echo 'phone 010-1234-5678' | tr 0-9 '[#*]'
phone ###-####-####
$ echo 'a,b;c' | tr ',;' '\t\t' | od -An -c
   a  \t   b  \t   c  \n

===== 11. tr -d, -s, -c
$ od -An -c crlf.txt
   a   l   p   h   a  \r  \n   b   e   t   a  \r  \n
$ tr -d '\r' < crlf.txt | od -An -c
   a   l   p   h   a  \n   b   e   t   a  \n
$ echo 'order 42 total 1,300' | tr -cd '0-9\n'
421300
$ echo 'aaa   bbb  ccc' | tr -s ' '
aaa bbb ccc
$ echo 'aaa   bbb  ccc' | tr -s ' ' '\n'
aaa
bbb
ccc
$ echo 'one  two
three' | tr -s '[:space:]' '\n'
one
two
three
$ echo 'hello' | tr -s 'l'
helo
$ echo 'a--b' | tr -d -- '-'
ab
