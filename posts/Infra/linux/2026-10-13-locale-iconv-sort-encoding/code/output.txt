== 1. 변수 우선순위 — LC_ALL > LC_* > LANG
-- LANG=ko_KR.UTF-8 만
LANG=ko_KR.UTF-8
LC_CTYPE="ko_KR.UTF-8"
LC_COLLATE="ko_KR.UTF-8"
LC_ALL=
-- LANG=ko_KR.UTF-8 + LC_COLLATE=C
LANG=ko_KR.UTF-8
LC_CTYPE="ko_KR.UTF-8"
LC_COLLATE="C"
LC_ALL=
-- 위에 LC_ALL=en_US.UTF-8 을 더하면
LANG=ko_KR.UTF-8
LC_CTYPE="en_US.UTF-8"
LC_COLLATE="en_US.UTF-8"
LC_ALL=en_US.UTF-8
