===== 1. 필드 번호를 먼저 확인한다
$ awk 'NR==1 {for (i = 1; i <= NF; i++) printf "%d:%s\n", i, $i}' access.log

2:-

5:"GET
6:/api/orders
7:HTTP/1.1"
8:200
9:41

===== 2. 기본 구분자는 연속 공백 — 상태코드는 $8, 응답시간은 $9
$ awk '{print $1, $6, $8, $9}' access.log | head -4
10.0.0.4 /api/orders 200 41
10.0.0.7 /api/stock 200 18
10.0.0.4 /api/orders 500 5002
10.0.0.9 /api/orders 200 39
$ awk 'END {print NR, NF}' access.log
10 9

===== 3. -F 로 구분자를 바꾼다
$ awk -F, 'NR>1 && NF {print $1, $3}' latency.csv
orders 41
"orders busan
stock 18
orders 5002
stock 3001
healthz 2

===== 4. --csv 는 따옴표 안의 구분자를 존중한다 (gawk 5.3+)
$ awk -F, 'NR==3 {print NF; print $1}' latency.csv
4
"orders
$ awk --csv 'NR==3 {print NF; print $1}' latency.csv
3
orders,legacy

===== 5. 패턴만 / 액션만 / 둘 다
$ awk '$8 >= 500' access.log
10.0.0.4 - - [22/Sep/2026:09:00:05] "POST /api/orders HTTP/1.1" 500 5002
10.0.0.9 - - [22/Sep/2026:09:00:12] "GET /api/stock HTTP/1.1" 503 3001
$ awk '$8 >= 500 {print $6, $8, $9}' access.log
/api/orders 500 5002
/api/stock 503 3001
$ awk '/POST/ {print $6, $8}' access.log
/api/orders 500
/api/orders 201
/api/orders 201

===== 6. 연관배열 — 상태코드별 건수
$ awk '{count[$8]++} END {for (code in count) print code, count[code]}' access.log
200 5
201 2
404 1
500 1
503 1

===== 7. for-in 순회 순서는 정의되어 있지 않다 — 정렬은 밖에서
$ awk '{count[$8]++} END {for (code in count) print code, count[code]}' access.log | sort -n
200 5
201 2
404 1
500 1
503 1

===== 8. 경로별 건수·합계·평균
$ awk '{n[$6]++; sum[$6]+=$9} END {for (p in n) printf "%-14s %3d %7d %9.1f\n", p, n[p], sum[p], sum[p]/n[p]}' access.log | sort
/api/orders      6    5268     878.0
/api/stock       2    3019    1509.5
/healthz         2       4       2.0

===== 9. OFS·ORS 로 출력 모양을 정한다
$ awk -v OFS=, '{print $1, $8}' access.log | head -3
10.0.0.4,200
10.0.0.7,200
10.0.0.4,500
$ awk -v ORS='|' '{print $8}' access.log; echo
200|200|500|200|200|201|503|201|404|200|

===== 10. $0 를 바꾸면 재분해되고, $1 을 바꾸면 $0 가 다시 조립된다
$ awk 'NR==1 {$1 = "HOST"; print}' access.log
HOST - - [22/Sep/2026:09:00:01] "GET /api/orders HTTP/1.1" 200 41
$ awk -v OFS=, 'NR==1 {$1 = $1; print}' access.log
10.0.0.4,-,-,[22/Sep/2026:09:00:01],"GET,/api/orders,HTTP/1.1",200,41

===== 11. -v 로 셸 값을 넘긴다
$ awk -v limit=500 '$8 >= limit {c++} END {print "임계 이상:", c+0}' access.log
임계 이상: 2

===== 12. 정의되지 않은 변수는 0이자 빈 문자열이다
$ awk 'END {print "[" undef "]", undef+0, length(undef)}' access.log
[] 0 0
$ awk '/NEVER_MATCH/ {c++} END {print "[" c "]", c+0}' access.log
[] 0

===== 13. NR·FNR·FILENAME — 파일이 여러 개일 때
$ awk '{print FILENAME, FNR, NR}' access.log latency.csv | sed -n '9,13p'
access.log 9 9
access.log 10 10
latency.csv 1 11
latency.csv 2 12
latency.csv 3 13

===== 14. 빈 줄도 레코드다
$ awk 'END {print NR}' latency.csv
8
$ awk 'NF {c++} END {print c}' latency.csv
7

===== 15. 숫자 비교와 문자열 비교가 갈린다
$ awk -F, 'NR>1 && NF && $3 > 100 {print $3}' latency.csv
busan
5002
3001
$ awk -F, 'NR>1 && NF && $3 "" > "100" {print $3}' latency.csv
41
busan
18
5002
3001
2

===== 16. -f 로 프로그램을 파일에서 읽는다
$ awk -f status_report.awk access.log
CLASS    CNT     MAX_MS
2xx        7         91
4xx        1          7
5xx        2       5002

===== 17. --posix 는 gawk 확장을 막는다
$ awk 'BEGIN {print gensub(/api/, "v2", "g", "/api/orders")}'
/v2/orders
$ awk --posix 'BEGIN {print gensub(/api/, "v2", "g", "/api/orders")}'

===== 18. 종료 코드는 exit 로 정한다
$ awk '$8 >= 500 {found=1} END {exit !found}' access.log; echo "종료 코드 $?"
종료 코드 0
$ awk '$8 >= 900 {found=1} END {exit !found}' access.log; echo "종료 코드 $?"
종료 코드 1
