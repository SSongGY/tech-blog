### 1. 개발 부서 4명 — 네 가지 EXCLUDE 가 프레임에 남기는 사람
SELECT emp_name, salary,
  group_concat(emp_name, ',') OVER (ORDER BY salary ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
                                    EXCLUDE NO OTHERS)  AS no_others,
  group_concat(emp_name, ',') OVER (ORDER BY salary ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
                                    EXCLUDE CURRENT ROW) AS current_row,
  group_concat(emp_name, ',') OVER (ORDER BY salary ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
                                    EXCLUDE GROUP)       AS grp,
  group_concat(emp_name, ',') OVER (ORDER BY salary ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
                                    EXCLUDE TIES)        AS ties
FROM employee
WHERE dept_name = '개발'
ORDER BY salary, emp_name
  emp_name | salary | no_others           | current_row    | grp            | ties
  ---------+--------+---------------------+----------------+----------------+--------------------
  가온     |    300 | 가온,나래,다온,라온 | 나래,다온,라온 | 나래,다온,라온 | 가온,나래,다온,라온
  나래     |    400 | 가온,나래,다온,라온 | 가온,다온,라온 | 가온,라온      | 가온,나래,라온
  다온     |    400 | 가온,나래,다온,라온 | 가온,나래,라온 | 가온,라온      | 가온,다온,라온

### 2. 나를 뺀 부서 평균 — EXCLUDE CURRENT ROW 와 (합-나)/(수-1)
SELECT dept_name, emp_name, salary,
  AVG(salary) OVER (PARTITION BY dept_name ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
                    EXCLUDE CURRENT ROW)                AS others_avg,
  (SUM(salary) OVER (PARTITION BY dept_name) - salary)
    / (COUNT(*) OVER (PARTITION BY dept_name) - 1)      AS by_formula,
  (SUM(salary) OVER (PARTITION BY dept_name) - salary) * 1.0
    / (COUNT(*) OVER (PARTITION BY dept_name) - 1)      AS by_formula_real
FROM employee
ORDER BY dept_name, salary, emp_name
  dept_name | emp_name | salary | others_avg        | by_formula | by_formula_real
  ----------+----------+--------+-------------------+------------+------------------
  개발      | 가온     |    300 | 433.3333333333333 |        433 | 433.3333333333333

### 3. 기본 프레임(RANGE) 그대로 — EXCLUDE CURRENT ROW 가 동점을 남긴다
SELECT emp_name, salary,
  SUM(salary) OVER (ORDER BY salary)                         AS running,
  SUM(salary) OVER (ORDER BY salary
                    RANGE UNBOUNDED PRECEDING
                    EXCLUDE CURRENT ROW)                     AS range_excl,
  SUM(salary) OVER (ORDER BY salary
                    ROWS UNBOUNDED PRECEDING
                    EXCLUDE CURRENT ROW)                     AS rows_excl,
  COUNT(*)    OVER (ORDER BY salary
                    RANGE UNBOUNDED PRECEDING
                    EXCLUDE CURRENT ROW)                     AS range_cnt,
  COUNT(*)    OVER (ORDER BY salary
                    ROWS UNBOUNDED PRECEDING
                    EXCLUDE CURRENT ROW)                     AS rows_cnt
FROM employee
WHERE dept_name = '개발'
ORDER BY salary, emp_name
  emp_name | salary | running | range_excl | rows_excl | range_cnt | rows_cnt
  ---------+--------+---------+------------+-----------+-----------+---------
  가온     |    300 |     300 |       NULL |      NULL |         0 |        0
  나래     |    400 |    1100 |        700 |       300 |         2 |        1
  다온     |    400 |    1100 |        700 |       700 |         2 |        2
  라온     |    500 |    1600 |       1100 |      1100 |         3 |        3

### 4. ORDER BY 가 없으면 — 파티션 전체가 동점 묶음이다
SELECT dept_name, emp_name,

  COUNT(*) OVER (PARTITION BY dept_name ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING EXCLUDE GROUP)       AS excl_grp,

### 5. 앞뒤 한 명씩만 — 이웃 평균과 내 값 비교
SELECT emp_name, salary,
  AVG(salary) OVER (ORDER BY salary, emp_name
                    ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING
                    EXCLUDE CURRENT ROW)                    AS neighbor_avg,
  group_concat(emp_name, ',') OVER (ORDER BY salary, emp_name
                    ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING
                    EXCLUDE CURRENT ROW)                    AS neighbors
FROM employee
ORDER BY salary, emp_name
  emp_name | salary | neighbor_avg | neighbors
  ---------+--------+--------------+----------
  가온     |    300 |        350.0 | 마루
  마루     |    350 |        340.0 | 가온,사름
  사름     |    380 |        375.0 | 마루,나래
  나래     |    400 |        390.0 | 사름,다온
  다온     |    400 |        425.0 | 나래,바름

### 6. 프레임을 쓰지 않는 함수에 EXCLUDE 를 붙이면
SELECT emp_name, salary,
  ROW_NUMBER()     OVER (ORDER BY salary ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING EXCLUDE CURRENT ROW) AS rn,
  RANK()           OVER (ORDER BY salary ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING EXCLUDE CURRENT ROW) AS rnk,
  FIRST_VALUE(emp_name)
                   OVER (ORDER BY salary ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING EXCLUDE CURRENT ROW) AS first_other
FROM employee
WHERE dept_name = '개발'
ORDER BY salary, emp_name
  emp_name | salary | rn | rnk | first_other
  ---------+--------+----+-----+------------
  가온     |    300 |  1 |   1 | 나래
  나래     |    400 |  2 |   2 | 가온
  다온     |    400 |  3 |   2 | 가온
  라온     |    500 |  4 |   4 | 가온

### 7. 문법 — EXCLUDE 는 프레임 경계 뒤에만 온다
  ORDER BY 뒤에 바로   실패   — ORDER BY salary EXCLUDE TIES
                    OperationalError: near "EXCLUDE": syntax error
  경계 앞에            실패   — ORDER BY salary EXCLUDE TIES ROWS UNBOUNDED PRECEDING
                    OperationalError: near "EXCLUDE": syntax error
  경계 뒤에            OK     — ORDER BY salary ROWS UNBOUNDED PRECEDING EXCLUDE TIES
