=== 1. INNER JOIN 과 LEFT JOIN 의 행 수 ===

[1-A INNER JOIN]
  SELECT c.name, o.id, o.status FROM customer AS c INNER JOIN sale_order AS o ON o.customer_id = c.id ORDER BY c.id, o.id
  ('김서준', 101, '완료')
  ('김서준', 102, '완료')
  ('김서준', 103, '취소')
  ('이하윤', 104, '취소')
  ('박도윤', 105, '완료')
  -> 5행

[1-B LEFT JOIN]
  SELECT c.name, o.id, o.status FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id ORDER BY c.id, o.id
  ('김서준', 101, '완료')
  ('김서준', 102, '완료')
  ('김서준', 103, '취소')
  ('이하윤', 104, '취소')
  ('박도윤', 105, '완료')
  ('최시우', None, None)
  -> 6행

=== 2. '완료' 주문만 붙이고 싶을 때 ===

[2-A 조건을 WHERE 에]
  SELECT c.name, o.id, o.status FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id WHERE o.status = '완료' ORDER BY c.id, o.id
  ('김서준', 101, '완료')
  ('김서준', 102, '완료')
  ('박도윤', 105, '완료')
  -> 3행

[2-B 조건을 ON 에]
  SELECT c.name, o.id, o.status FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id AND o.status = '완료' ORDER BY c.id, o.id
  ('김서준', 101, '완료')
  ('김서준', 102, '완료')
  ('이하윤', None, None)
  ('박도윤', 105, '완료')
  ('최시우', None, None)
  -> 5행

[2-C WHERE 에 OR IS NULL 을 덧붙이면]
  SELECT c.name, o.id, o.status FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id WHERE o.status = '완료' OR o.id IS NULL ORDER BY c.id, o.id
  ('김서준', 101, '완료')
  ('김서준', 102, '완료')
  ('박도윤', 105, '완료')
  ('최시우', None, None)
  -> 4행

=== 3. 고객별 완료 주문 수 ===

[3-A COUNT(*)]
  SELECT c.name, COUNT(*) FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id AND o.status = '완료' GROUP BY c.id ORDER BY c.id
  ('김서준', 2)
  ('이하윤', 1)
  ('박도윤', 1)
  ('최시우', 1)
  -> 4행

[3-B COUNT(o.id)]
  SELECT c.name, COUNT(o.id) FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id AND o.status = '완료' GROUP BY c.id ORDER BY c.id
  ('김서준', 2)
  ('이하윤', 0)
  ('박도윤', 1)
  ('최시우', 0)
  -> 4행

[3-C SUM 은 0 이 아니라 NULL]
  SELECT c.name, SUM(o.amount), COALESCE(SUM(o.amount), 0) FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id AND o.status = '완료' GROUP BY c.id ORDER BY c.id
  ('김서준', 200, 200)
  ('이하윤', None, 0)
  ('박도윤', 90, 90)
  ('최시우', None, 0)
  -> 4행

=== 4. 주문이 없는 고객 찾기 ===

[4-A 키 컬럼으로 IS NULL]
  SELECT c.name FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id WHERE o.id IS NULL
  ('최시우',)
  -> 1행

[4-B 비어 있을 수 있는 컬럼으로 IS NULL]
  SELECT c.name, o.id FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id WHERE o.coupon_code IS NULL ORDER BY c.id, o.id
  ('김서준', 102)
  ('김서준', 103)
  ('박도윤', 105)
  ('최시우', None)
  -> 4행

[4-C NOT EXISTS]
  SELECT c.name FROM customer AS c WHERE NOT EXISTS (SELECT 1 FROM sale_order AS o WHERE o.customer_id = c.id)
  ('최시우',)
  -> 1행

=== 5. 왼쪽 표 조건을 ON 에 쓰면 ===

[5-A 왼쪽 표 조건을 WHERE 에]
  SELECT c.name, o.id FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id WHERE c.id = 1 ORDER BY c.id, o.id

  ('김서준', 102)
  ('김서준', 103)
  -> 3행

[5-B 왼쪽 표 조건을 ON 에]
  SELECT c.name, o.id FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id AND c.id = 1 ORDER BY c.id, o.id
  ('김서준', 101)
  ('김서준', 102)
  ('김서준', 103)
  ('이하윤', None)
  ('박도윤', None)
  ('최시우', None)
  -> 6행

=== 6. LEFT JOIN 뒤에 INNER JOIN 을 이으면 ===

[6-A LEFT JOIN 만]
  SELECT c.name, o.id FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id AND o.status = '완료' ORDER BY c.id, o.id
  ('김서준', 101)
  ('김서준', 102)
  ('이하윤', None)
  ('박도윤', 105)
  ('최시우', None)
  -> 5행

[6-B 이어서 INNER JOIN shipment]
  SELECT c.name, o.id, s.id FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id AND o.status = '완료' INNER JOIN shipment AS s ON s.order_id = o.id ORDER BY c.id, o.id
  ('김서준', 101, 301)
  ('박도윤', 105, 302)
  -> 2행

[6-C 이어서 LEFT JOIN shipment]
  SELECT c.name, o.id, s.id FROM customer AS c LEFT JOIN sale_order AS o ON o.customer_id = c.id AND o.status = '완료' LEFT JOIN shipment AS s ON s.order_id = o.id ORDER BY c.id, o.id
  ('김서준', 101, 301)
  ('김서준', 102, None)
  ('이하윤', None, None)
  ('박도윤', 105, 302)
  ('최시우', None, None)
  -> 5행
