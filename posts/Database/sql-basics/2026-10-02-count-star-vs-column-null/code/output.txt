== 1. 같은 표, 세 가지 COUNT ==

[1-A 행 수 / 휴대폰 수 / 지역 수 / 서로 다른 지역 수]
  SELECT COUNT(*), COUNT(mobile), COUNT(city), COUNT(DISTINCT city) FROM member
  ('COUNT(*)', 'COUNT(mobile)', 'COUNT(city)', 'COUNT(DISTINCT city)')
  (5, 3, 4, 2)

[1-B 빈 문자열은 NULL이 아니므로 센다]
  SELECT COUNT(mobile), COUNT(NULLIF(mobile, '')) FROM member
  ('COUNT(mobile)', "COUNT(NULLIF(mobile, ''))")
  (3, 2)

[1-C 괄호 안에 상수를 넣으면]
  SELECT COUNT(1), COUNT('x'), COUNT(NULL) FROM member
  ('COUNT(1)', "COUNT('x')", 'COUNT(NULL)')
  (5, 5, 0)

[1-D SELECT DISTINCT 로 뽑은 지역을 세면]
  SELECT COUNT(*) FROM (SELECT DISTINCT city FROM member)
  ('COUNT(*)',)
  (3,)

== 2. 세는 대상이 하나도 없을 때 ==

[2-A 조건에 맞는 행이 없으면 COUNT는 0, SUM은 NULL]
  SELECT COUNT(*), COUNT(mobile), SUM(id) FROM member WHERE city = '제주'
  ('COUNT(*)', 'COUNT(mobile)', 'SUM(id)')
  (0, 0, None)

== 3. GROUP BY 와 함께 ==

[3-A 지역별 회원 수와 휴대폰 등록 수]
  SELECT city, COUNT(*) AS member_cnt, COUNT(mobile) AS mobile_cnt FROM member GROUP BY city ORDER BY city
  ('city', 'member_cnt', 'mobile_cnt')
  (None, 1, 1)
  ('부산', 1, 0)
  ('서울', 3, 2)

== 4. LEFT JOIN 뒤에 무엇을 세는가 ==

[4-A 회원별 주문 수를 COUNT(*)로]
  SELECT m.id, m.name, COUNT(*) AS order_cnt FROM member m LEFT JOIN shop_order o ON o.member_id = m.id GROUP BY m.id ORDER BY m.id
  ('id', 'name', 'order_cnt')
  (1, '김도윤', 2)
  (2, '이서준', 1)
  (3, '박하은', 2)
  (4, '최유나', 1)
  (5, '정민호', 1)

[4-B 회원별 주문 수를 COUNT(o.id)로]
  SELECT m.id, m.name, COUNT(o.id) AS order_cnt FROM member m LEFT JOIN shop_order o ON o.member_id = m.id GROUP BY m.id ORDER BY m.id
  ('id', 'name', 'order_cnt')
  (1, '김도윤', 2)
  (2, '이서준', 0)
  (3, '박하은', 2)
  (4, '최유나', 0)
  (5, '정민호', 1)

[4-C 조인 직후의 행 — 주문 없는 회원도 한 줄은 남는다]
  SELECT m.id, o.id AS order_id FROM member m LEFT JOIN shop_order o ON o.member_id = m.id WHERE m.id IN (2, 4) ORDER BY m.id
  ('id', 'order_id')
  (2, None)
  (4, None)

== 5. 조건부로 세기 ==

[5-A CASE 로 조건에 맞을 때만 값을 준다]
  SELECT COUNT(CASE WHEN city = '서울' THEN 1 END) AS seoul_cnt, COUNT(CASE WHEN city = '서울' THEN 1 ELSE 0 END) AS wrong_cnt FROM member
  ('seoul_cnt', 'wrong_cnt')
  (3, 5)

[5-B FILTER 절]
  SELECT COUNT(*) FILTER (WHERE city = '서울') AS seoul_cnt FROM member
  ('seoul_cnt',)
  (3,)

[5-C DISTINCT 에 컬럼 두 개]
  SELECT COUNT(DISTINCT city, mobile) FROM member
  에러: OperationalError: wrong number of arguments to function COUNT()
