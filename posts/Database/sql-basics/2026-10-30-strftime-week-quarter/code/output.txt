1. 포맷 문자 하나하나가 무엇을 내놓는가
================================================================
SELECT strftime('%Y', t) AS "%Y",
               strftime('%m', t) AS "%m",
               strftime('%d', t) AS "%d",
               strftime('%j', t) AS "%j",
               strftime('%H:%M:%S', t) AS "%H:%M:%S",
               strftime('%s', t) AS "%s",
               strftime('%J', t) AS "%J"
        FROM (SELECT '2026-01-03 14:05:06' AS t)
----------------------------------------------------------------
  %Y   | %m | %d | %j  | %H:%M:%S | %s         | %J            
  -----+----+----+-----+----------+------------+---------------
  2026 | 01 | 03 | 003 | 14:05:06 | 1767449106 | 2461044.086875

SELECT strftime('%w', t) AS "%w",
               strftime('%u', t) AS "%u",
               strftime('%U', t) AS "%U",
               strftime('%V', t) AS "%V",
               strftime('%W', t) AS "%W",
               strftime('%G', t) AS "%G",
               strftime('%g', t) AS "%g"
        FROM (SELECT '2026-01-03 14:05:06' AS t)
----------------------------------------------------------------
  %w | %u | %U | %V | %W | %G   | %g
  ---+----+----+----+----+------+---
  6  | 6  | 00 | 01 | 00 | 2026 | 26

SELECT strftime('%a', t) AS "%a",

2. 연초 경계 -- %U %V %W %G 가 서로 다른 답을 낸다
================================================================
SELECT ordered,
               strftime('%u', ordered) AS iso요일,
               strftime('%Y', ordered) AS 달력연도,
               strftime('%U', ordered) AS "%U",
               strftime('%V', ordered) AS "%V",
               strftime('%W', ordered) AS "%W",
               strftime('%G', ordered) AS "%G"
        FROM orders
        WHERE ordered BETWEEN '2025-12-28' AND '2026-01-05'
        ORDER BY ordered
----------------------------------------------------------------
  ordered    | iso요일 | 달력연도 | %U | %V | %W | %G  
  -----------+-------+------+----+----+----+-----
  2025-12-28 | 7     | 2025 | 52 | 52 | 51 | 2025
  2025-12-29 | 1     | 2025 | 52 | 01 | 52 | 2026
  2025-12-31 | 3     | 2025 | 52 | 01 | 52 | 2026
  2026-01-01 | 4     | 2026 | 00 | 01 | 00 | 2026
  2026-01-03 | 6     | 2026 | 00 | 01 | 00 | 2026
  2026-01-04 | 7     | 2026 | 01 | 01 | 00 | 2026
  2026-01-05 | 1     | 2026 | 01 | 02 | 01 | 2026

3. 분기 -- 전용 포맷 문자가 없으므로 월에서 계산한다
================================================================
SELECT ordered,
               CAST(strftime('%m', ordered) AS INTEGER) AS 월,
               (CAST(strftime('%m', ordered) AS INTEGER) + 2) / 3 AS 분기
        FROM orders
        ORDER BY ordered
----------------------------------------------------------------
  ordered    | 월  | 분기
  -----------+----+---
  2025-12-28 | 12 | 4 
  2025-12-29 | 12 | 4 
  2025-12-31 | 12 | 4 
  2026-01-01 | 1  | 1 
  2026-01-03 | 1  | 1 
  2026-01-04 | 1  | 1 
  2026-01-05 | 1  | 1 
  2026-02-14 | 2  | 1 
  2026-04-01 | 4  | 2 
  2026-07-31 | 7  | 3 
  2026-10-30 | 10 | 4 
  2026-12-31 | 12 | 4 

SELECT strftime('%Y', ordered) || '-Q'
                 || (CAST(strftime('%m', ordered) AS INTEGER) + 2) / 3 AS 분기,
               COUNT(*) AS 건수,
               SUM(amount) AS 합계
        FROM orders
        GROUP BY 분기
        ORDER BY 분기
----------------------------------------------------------------
  분기  | 건수 | 합계    
  ----+----+-------
  675 | 12 | 203000

4. 주 단위 집계 -- 주의 시작일로 묶는 쪽이 안전하다
================================================================
SELECT strftime('%G-W%V', ordered) AS iso주,
               date(ordered, '-6 days', 'weekday 1') AS 주시작,
               COUNT(*) AS 건수,
               SUM(amount) AS 합계
        FROM orders
        WHERE ordered < '2026-02-01'
        GROUP BY iso주
        ORDER BY iso주
----------------------------------------------------------------
  iso주     | 주시작        | 건수 | 합계   
  ---------+------------+----+------
  2025-W52 | 2025-12-22 | 1  | 12000
  2026-W01 | 2025-12-29 | 5  | 59000
  2026-W02 | 2026-01-05 | 1  | 11000

SELECT ordered,
               date(ordered, '-6 days', 'weekday 1') AS 월요일시작,

5. 월초·월말과 +N months
================================================================
SELECT ordered,
               date(ordered, 'start of month') AS 월초,
               date(ordered, 'start of month', '+1 month', '-1 day') AS 월말
        FROM orders
        WHERE ordered IN ('2026-02-14', '2026-07-31', '2026-12-31')
        ORDER BY ordered
----------------------------------------------------------------
  ordered    | 월초         | 월말        
  -----------+------------+-----------
  2026-02-14 | 2026-02-01 | 2026-02-28
  2026-07-31 | 2026-07-01 | 2026-07-31
  2026-12-31 | 2026-12-01 | 2026-12-31

SELECT d,
               date(d, '+1 month') AS 기본값,
               date(d, '+1 month', 'floor') AS floor,
               date(d, '+1 month', 'ceiling') AS ceiling
        FROM (SELECT '2026-01-31' AS d
              UNION ALL SELECT '2026-03-31'
              UNION ALL SELECT '2024-01-29')
        ORDER BY d
----------------------------------------------------------------
  d          | 기본값        | floor      | ceiling   
  -----------+------------+------------+-----------
  2024-01-29 | 2024-02-29 | 2024-02-29 | 2024-02-29
  2026-01-31 | 2026-03-03 | 2026-02-28 | 2026-03-03
  2026-03-31 | 2026-05-01 | 2026-04-30 | 2026-05-01

7. 반환값은 문자열이다 -- 숫자처럼 쓰면 비교가 틀어진다
================================================================
SELECT typeof(strftime('%m', '2026-10-30'))        AS 반환타입,
               strftime('%m', '2026-01-30') > 9            AS "'01' > 9",
               CAST(strftime('%m', '2026-01-30') AS INTEGER) > 9 AS "CAST 후",
               '2026-Q1' / 3                               AS "'2026-Q1' / 3",
               CAST('2026-Q1' AS INTEGER)                  AS "CAST('2026-Q1')"
----------------------------------------------------------------
  반환타입 | '01' > 9 | CAST 후 | '2026-Q1' / 3 | CAST('2026-Q1')
  -----+----------+--------+---------------+----------------
  text | 1        | 0      | 675           | 2026
