== 1) 중앙값 함수가 있는가 ==
  sql> SELECT median(salary) FROM employee
  OperationalError: no such function: median

  sql> SELECT percentile_cont(salary, 0.5) FROM employee
  OperationalError: no such function: percentile_cont

== 2) 평균과 중앙값 — 비교용 정답은 파이썬 statistics.median ==
  sql> SELECT dept, count(salary) AS n, avg(salary) AS mean FROM employee GROUP BY dept ORDER BY dept
  dept | n | mean
  'dev' | 4 | 4650.25
  'ops' | 2 | 4100.0
  'sales' | 5 | 5260.0

  python median dev   [4200, 4500, 4601, 5300] -> 4550.5
  python median ops   [3900, 4300] -> 4100.0
  python median sales [3200, 3500, 3500, 4100, 12000] -> 3500

== 3) 전체 중앙값 — 가운데 번호 lo, hi ==
  sql> SELECT salary, ROW_NUMBER() OVER (ORDER BY salary) AS rn, COUNT(*) OVER () AS cnt, (COUNT(*) OVER () + 1) / 2 AS lo, (COUNT(*) OVER () + 2) / 2 AS hi FROM employee WHERE salary IS NOT NULL
  salary | rn | cnt | lo | hi
  3200 | 1 | 11 | 6 | 6
  3500 | 2 | 11 | 6 | 6
  3500 | 3 | 11 | 6 | 6
  3900 | 4 | 11 | 6 | 6
  4100 | 5 | 11 | 6 | 6
  4200 | 6 | 11 | 6 | 6
  4300 | 7 | 11 | 6 | 6
  4500 | 8 | 11 | 6 | 6
  4601 | 9 | 11 | 6 | 6
  5300 | 10 | 11 | 6 | 6
  12000 | 11 | 11 | 6 | 6

  sql> WITH ordered AS ( SELECT salary, ROW_NUMBER() OVER (ORDER BY salary) AS rn, COUNT(*) OVER () AS cnt, (COUNT(*) OVER () + 1) / 2 AS lo, (COUNT(*) OVER () + 2) / 2 AS hi FROM employee WHERE salary IS NOT NULL ) SELECT avg(salary) AS median FROM ordered WHERE rn IN (lo, hi)
  median
  4200.0

  python median 전체 11개 -> 4200

== 4) 부서별 중앙값 ==
  sql> WITH ordered AS ( SELECT dept, salary, ROW_NUMBER() OVER (PARTITION BY dept ORDER BY salary) AS rn, COUNT(*) OVER (PARTITION BY dept) AS cnt FROM employee WHERE salary IS NOT NULL ) SELECT dept, cnt, avg(salary) AS median FROM ordered WHERE rn IN ((cnt + 1) / 2, (cnt + 2) / 2) GROUP BY dept ORDER BY dept
  dept | cnt | median
  'dev' | 4 | 4550.5
  'ops' | 2 | 4100.0
  'sales' | 5 | 3500.0

== 5-a) WHERE salary IS NOT NULL 을 빼면 — ops 부서 ==
  sql> SELECT name, salary, ROW_NUMBER() OVER (ORDER BY salary) AS rn, COUNT(*) OVER () AS cnt FROM employee WHERE dept = 'ops'
  name | salary | rn | cnt
  'oh' | NULL | 1 | 3
  'han' | 3900 | 2 | 3
  'seo' | 4300 | 3 | 3

  sql> WITH ordered AS ( SELECT salary, ROW_NUMBER() OVER (ORDER BY salary) AS rn, COUNT(*) OVER () AS cnt FROM employee WHERE dept = 'ops') SELECT avg(salary) AS median FROM ordered WHERE rn IN ((cnt + 1) / 2, (cnt + 2) / 2)
  median
  3900.0

== 5-b) ROW_NUMBER 대신 RANK — sales 부서, 3500 이 둘 ==
  sql> SELECT name, salary, RANK() OVER (ORDER BY salary) AS rk, ROW_NUMBER() OVER (ORDER BY salary) AS rn FROM employee WHERE dept = 'sales'
  name | salary | rk | rn
  'kim' | 3200 | 1 | 1
  'lee' | 3500 | 2 | 2
  'park' | 3500 | 2 | 3
  'choi' | 4100 | 4 | 4
  'jung' | 12000 | 5 | 5

  sql> WITH ordered AS ( SELECT salary, RANK() OVER (ORDER BY salary) AS rk, COUNT(*) OVER () AS cnt FROM employee WHERE dept = 'sales' AND salary <> 3200) SELECT count(*) AS hit_rows, avg(salary) AS median FROM ordered WHERE rk IN ((cnt + 1) / 2, (cnt + 2) / 2)
  hit_rows | median
  1 | 4100.0

== 5-c) 두 값을 더해 2 로 나누면 — dev 부서 4500, 4601 ==
  sql> SELECT (4500 + 4601) / 2 AS int_div, (4500 + 4601) / 2.0 AS real_div, avg(x) AS by_avg FROM (SELECT 4500 AS x UNION ALL SELECT 4601)
  int_div | real_div | by_avg
  4550 | 4550.5 | 4550.5

== 6) 창 함수 없이 LIMIT/OFFSET ==
  sql> SELECT avg(salary) AS median FROM ( SELECT salary FROM employee WHERE salary IS NOT NULL ORDER BY salary LIMIT 2 - (SELECT count(salary) FROM employee) % 2 OFFSET (SELECT (count(salary) - 1) / 2 FROM employee) )
  median
  4200.0

  -- lim 을 지워 10개(짝수)로 만든 뒤 두 방법을 다시 비교
  sql> SELECT avg(salary) AS median FROM ( SELECT salary FROM employee WHERE salary IS NOT NULL ORDER BY salary LIMIT 2 - (SELECT count(salary) FROM employee) % 2 OFFSET (SELECT (count(salary) - 1) / 2 FROM employee) )
  median
  4150.0
