== 1. 문자열로 저장한 날짜의 정렬과 비교 ==

[1-A ORDER BY ordered_at — 글자 순서로 정렬된다]
  SELECT id, ordered_at FROM orders ORDER BY ordered_at
  (1, '2026-09-05 09:30:00')
  (6, '2026-09-30')
  (5, '2026-09-30 23:59:59')
  (4, '2026-10-01 08:00:00')
  (2, '2026-9-5 14:00:00')
  (3, '2026/09/28 18:10:00')
  -> 6행

[1-B 9월 주문을 BETWEEN 으로 찾기]
  SELECT id, ordered_at FROM orders WHERE ordered_at BETWEEN '2026-09-01' AND '2026-09-30'
  (1, '2026-09-05 09:30:00')
  (6, '2026-09-30')
  -> 2행

[1-C 날짜 함수가 형식을 알아보는가]
  SELECT id, ordered_at, date(ordered_at) FROM orders ORDER BY id
  (1, '2026-09-05 09:30:00', '2026-09-05')
  (2, '2026-9-5 14:00:00', None)
  (3, '2026/09/28 18:10:00', None)
  (4, '2026-10-01 08:00:00', '2026-10-01')
  (5, '2026-09-30 23:59:59', '2026-09-30')
  (6, '2026-09-30', '2026-09-30')
  -> 6행

[1-D 형식 검사 — 알아보지 못한 행 찾기]
  SELECT id, ordered_at FROM orders WHERE datetime(ordered_at) IS NULL
  (2, '2026-9-5 14:00:00')
  (3, '2026/09/28 18:10:00')
  -> 2행

== 2. ISO 8601 로 고친 뒤 ==

[2-A 다시 정렬]
  SELECT id, ordered_at FROM orders ORDER BY ordered_at
  (1, '2026-09-05 09:30:00')
  (2, '2026-09-05 14:00:00')
  (3, '2026-09-28 18:10:00')
  (6, '2026-09-30 00:00:00')
  (5, '2026-09-30 23:59:59')
  (4, '2026-10-01 08:00:00')
  -> 6행

[2-B BETWEEN 끝값에 날짜만 쓰면]
  SELECT id, ordered_at FROM orders WHERE ordered_at BETWEEN '2026-09-01' AND '2026-09-30'
  (1, '2026-09-05 09:30:00')
  (2, '2026-09-05 14:00:00')
  (3, '2026-09-28 18:10:00')
  -> 3행

[2-C 반열림 구간 — 이상 AND 미만]
  SELECT id, ordered_at FROM orders WHERE ordered_at >= '2026-09-01' AND ordered_at < '2026-10-01'
  (1, '2026-09-05 09:30:00')
  (2, '2026-09-05 14:00:00')
  (3, '2026-09-28 18:10:00')
  (5, '2026-09-30 23:59:59')
  (6, '2026-09-30 00:00:00')
  -> 5행

== 3. 같은 순간을 세 가지 형태로 ==

[3-A TEXT · REAL · INTEGER]
  SELECT '2026-10-01 08:00:00' AS iso_text, julianday('2026-10-01 08:00:00') AS julian_day, unixepoch('2026-10-01 08:00:00') AS unix_time
  ('2026-10-01 08:00:00', 2461314.8333333335, 1790841600)
  -> 1행

[3-B 숫자에서 다시 글자로]
  WITH t(jd, ux) AS ( SELECT julianday('2026-10-01 08:00:00'), unixepoch('2026-10-01 08:00:00') ) SELECT datetime(jd), datetime(ux, 'unixepoch'), datetime(ux), datetime(ux, 'auto') FROM t
  ('2026-10-01 08:00:00', '2026-10-01 08:00:00', None, '2026-10-01 08:00:00')
  -> 1행

[3-C typeof — 세 값의 저장 부류]
  SELECT typeof('2026-10-01'), typeof(julianday('2026-10-01')), typeof(unixepoch('2026-10-01'))
  ('text', 'real', 'integer')
  -> 1행

== 4. 더하고 빼기 — 수식어 ==

[4-A 일·시간·분 더하기]
  SELECT datetime('2026-09-30 23:59:59', '+1 second'), date('2026-09-30', '+7 days'), datetime('2026-09-30 22:00:00', '+3 hours')
  ('2026-10-01 00:00:00', '2026-10-07', '2026-10-01 01:00:00')
  -> 1행

[4-B 월말에 한 달 더하기]
  SELECT date('2026-01-31', '+1 month'), date('2026-01-31', 'start of month', '+1 month', '-1 day'), date('2026-01-31', '+1 month', 'floor')
  ('2026-03-03', '2026-01-31', '2026-02-28')
  -> 1행

[4-C 수식어는 왼쪽부터 차례로 적용된다]
  SELECT date('2026-01-31', 'start of month', '+1 month'), date('2026-01-31', '+1 month', 'start of month')
  ('2026-02-01', '2026-03-01')
  -> 1행

[4-D 이번 달 첫날·말일, 다음 월요일]
  SELECT date('2026-10-01 08:00:00', 'start of month'), date('2026-10-01 08:00:00', 'start of month', '+1 month', '-1 day'), date('2026-10-01', 'weekday 1')
  ('2026-10-01', '2026-10-31', '2026-10-05')
  -> 1행

[4-E 없는 날짜를 넣으면]
  SELECT date('2026-02-30'), date('2026-02-31'), date('2026-02-32'), date('2026-13-01')
  ('2026-03-02', '2026-03-03', None, None)
  -> 1행

== 5. 두 시각의 차이 ==

[5-A 문자열끼리 빼면]
  SELECT '2026-10-01' - '2026-09-05'
  (0,)
  -> 1행

[5-B julianday 차이(일)와 unixepoch 차이(초)]
  SELECT julianday('2026-10-01') - julianday('2026-09-05'), unixepoch('2026-10-01 08:00:00') - unixepoch('2026-09-30 23:59:59')
  (26.0, 28801)
  -> 1행

[5-C timediff — 년·월·일로 풀어 쓴 차이]
  SELECT timediff('2026-10-01', '2026-09-05'), timediff('2026-03-01', '2026-01-31')
  ('+0000-00-26 00:00:00.000', '+0000-00-29 00:00:00.000')
  -> 1행

[5-D 주문마다 마감(10-01 08:00)까지 남은 시간]
  SELECT id, ordered_at, ROUND((julianday('2026-10-01 08:00:00') - julianday(ordered_at)) * 24, 2) AS hours_left FROM orders ORDER BY ordered_at
  (1, '2026-09-05 09:30:00', 622.5)
  (2, '2026-09-05 14:00:00', 618.0)
  (3, '2026-09-28 18:10:00', 61.83)
  (6, '2026-09-30 00:00:00', 32.0)
  (5, '2026-09-30 23:59:59', 8.0)
  (4, '2026-10-01 08:00:00', 0.0)
  -> 6행

== 6. 잘라서 묶기 — strftime ==

[6-A 형식 지정자]
  SELECT strftime('%Y', '2026-10-01 08:05:09'), strftime('%m/%d %H:%M', '2026-10-01 08:05:09'), strftime('%w', '2026-10-01'), strftime('%j', '2026-10-01')
  ('2026', '10/01 08:05', '4', '274')
  -> 1행

[6-B 날짜별 주문 수]
  SELECT date(ordered_at) AS order_date, COUNT(*) AS order_count FROM orders GROUP BY date(ordered_at) ORDER BY order_date
  ('2026-09-05', 2)
  ('2026-09-28', 1)
  ('2026-09-30', 2)
  ('2026-10-01', 1)
  -> 4행

[6-C strftime 결과의 타입]
  SELECT strftime('%m', '2026-09-05') = 9, strftime('%m', '2026-09-05') = '09', CAST(strftime('%m', '2026-09-05') AS INTEGER) = 9
  (0, 1, 1)
  -> 1행

== 7. 시간대 ==

[7-A 시간대 표기가 붙은 값은 UTC 로 바꿔 계산한다]
  SELECT datetime('2026-10-01 08:00:00+09:00'), datetime('2026-10-01T08:00:00Z')
  ('2026-09-30 23:00:00', '2026-10-01 08:00:00')
  -> 1행

[7-B 같은 순간인데 글자는 다르다]
  SELECT '2026-10-01 08:00:00+09:00' = '2026-09-30 23:00:00', datetime('2026-10-01 08:00:00+09:00') = '2026-09-30 23:00:00'
  (0, 1)
  -> 1행
