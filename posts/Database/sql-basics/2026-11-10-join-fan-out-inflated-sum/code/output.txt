  sql> SELECT c.name, (SELECT COUNT(*) FROM orders o WHERE o.customer_id = c.customer_id) AS m, (SELECT COUNT(*) FROM payment p WHERE p.customer_id = c.customer_id) AS n, COUNT(*) AS joined_rows FROM customer c LEFT JOIN orders o ON o.customer_id = c.customer_id LEFT JOIN payment p ON p.customer_id = c.customer_id GROUP BY c.customer_id ORDER BY c.customer_id

== 2) 그대로 집계하면 ==
  sql> SELECT c.name, COUNT(o.order_id) AS order_count, SUM(o.amount) AS order_total, SUM(p.paid) AS paid_total FROM customer c LEFT JOIN orders o ON o.customer_id = c.customer_id LEFT JOIN payment p ON p.customer_id = c.customer_id GROUP BY c.customer_id ORDER BY c.customer_id
  name | order_count | order_total | paid_total

== 3) DISTINCT 로 막으려 하면 ==
  sql> SELECT c.name, COUNT(DISTINCT o.order_id) AS order_count, SUM(DISTINCT o.amount) AS order_total, SUM(DISTINCT p.paid) AS paid_total FROM customer c LEFT JOIN orders o ON o.customer_id = c.customer_id LEFT JOIN payment p ON p.customer_id = c.customer_id GROUP BY c.customer_id ORDER BY c.customer_id
  name | order_count | order_total | paid_total
  kim | 3 | 15000 | 25000
  lee | 2 | 10000 | NULL
  park | 1 | 20000 | 20000

== 4) 먼저 집계하고 조인 ==
  sql> SELECT c.name, o.order_total, p.paid_total FROM customer c LEFT JOIN (SELECT customer_id, SUM(amount) AS order_total FROM orders GROUP BY customer_id) o ON o.customer_id = c.customer_id LEFT JOIN (SELECT customer_id, SUM(paid) AS paid_total FROM payment GROUP BY customer_id) p ON p.customer_id = c.customer_id ORDER BY c.customer_id
  name | order_total | paid_total
  kim | 25000 | 25000
  lee | 10000 | NULL
  park | 20000 | 20000

== 5) 상관 부질의 ==
  sql> SELECT c.name, (SELECT SUM(amount) FROM orders o WHERE o.customer_id = c.customer_id) AS order_total, (SELECT SUM(paid) FROM payment p WHERE p.customer_id = c.customer_id) AS paid_total FROM customer c ORDER BY c.customer_id
  name | order_total | paid_total
  kim | 25000 | 25000
  lee | 10000 | NULL
  park | 20000 | 20000

== 6) 실행계획 ==
  sql> SELECT c.name, o.order_total, p.paid_total FROM customer c LEFT JOIN (SELECT customer_id, SUM(amount) AS order_total FROM orders GROUP BY customer_id) o ON o.customer_id = c.customer_id LEFT JOIN (SELECT customer_id, SUM(paid) AS paid_total FROM payment GROUP BY customer_id) p ON p.customer_id = c.customer_id ORDER BY c.customer_id
  QUERY PLAN
  |--MATERIALIZE o
  |  |--SCAN orders
  |  `--USE TEMP B-TREE FOR GROUP BY
  |--MATERIALIZE p
  |  |--SCAN payment
  |  `--USE TEMP B-TREE FOR GROUP BY
  |--SCAN c
  |--BLOOM FILTER ON o (customer_id=?)
  |--SEARCH o USING AUTOMATIC COVERING INDEX (customer_id=?) LEFT-JOIN
  |--BLOOM FILTER ON p (customer_id=?)
  `--SEARCH p USING AUTOMATIC COVERING INDEX (customer_id=?) LEFT-JOIN

  sql> SELECT c.name, (SELECT SUM(amount) FROM orders o WHERE o.customer_id = c.customer_id) AS order_total, (SELECT SUM(paid) FROM payment p WHERE p.customer_id = c.customer_id) AS paid_total FROM customer c ORDER BY c.customer_id
  QUERY PLAN
  |--SCAN c
  |--CORRELATED SCALAR SUBQUERY 1
  |  `--SCAN o
  `--CORRELATED SCALAR SUBQUERY 2
     `--SCAN p
