[2] 경과 월 — 일수/30 과 달력 월 차이가 갈리는 주문만
  sql> SELECT o.customer_id, f.first_date, o.order_date, julianday(o.order_date) - julianday(f.first_date) AS days, CAST((julianday(o.order_date) - julianday(f.first_date)) / 30 AS INTEGER) AS by_days, (CAST(strftime('%Y', o.order_date) AS INTEGER) * 12 + CAST(strftime('%m', o.order_date) AS INTEGER)) - (CAST(strftime('%Y', f.first_date) AS INTEGER) * 12 + CAST(strftime('%m', f.first_date) AS INTEGER)) AS by_month FROM customer_order AS o JOIN ( SELECT customer_id, MIN(order_date) AS first_date FROM customer_order GROUP BY customer_id ) AS f USING (customer_id) WHERE CAST((julianday(o.order_date) - julianday(f.first_date)) / 30 AS INTEGER) <> (CAST(strftime('%Y', o.order_date) AS INTEGER) * 12 + CAST(strftime('%m', o.order_date) AS INTEGER)) - (CAST(strftime('%Y', f.first_date) AS INTEGER) * 12 + CAST(strftime('%m', f.first_date) AS INTEGER)) ORDER BY o.customer_id, o.order_date
    ('customer_id', 'first_date', 'order_date', 'days', 'by_days', 'by_month')

  sql> SELECT strftime('%Y-%m', f.first_date) AS cohort, (CAST(strftime('%Y', o.order_date) AS INTEGER) * 12 + CAST(strftime('%m', o.order_date) AS INTEGER)) - (CAST(strftime('%Y', f.first_date) AS INTEGER) * 12 + CAST(strftime('%m', f.first_date) AS INTEGER)) AS month_no, COUNT(*) AS order_rows, COUNT(DISTINCT o.customer_id) AS customers FROM customer_order AS o JOIN ( SELECT customer_id, MIN(order_date) AS first_date FROM customer_order GROUP BY customer_id ) AS f USING (customer_id) GROUP BY cohort, month_no ORDER BY cohort, month_no
