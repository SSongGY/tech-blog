1) 선언 타입 문자열 -> 친화성 (문서의 다섯 규칙을 위에서 아래로)
================================================================
  INT              -> INTEGER
  BIGINT           -> INTEGER
  VARCHAR(20)      -> TEXT
  TEXT             -> TEXT
  DECIMAL(10,2)    -> NUMERIC
  FLOAT            -> REAL
  DOUBLE           -> REAL
  BOOLEAN          -> NUMERIC
  DATE             -> NUMERIC
  FLOATING POINT   -> INTEGER
  STRING           -> NUMERIC
  BLOB             -> BLOB
  (선언 없음)          -> BLOB

================================================================
3) CAST 는 친화성과 다르다
================================================================

-- CAST 는 값을 강제로 바꾼다. 'abc' 도 0 이 된다
   SELECT CAST('abc' AS INTEGER), CAST('12abc' AS INTEGER), CAST(' 12 ' AS INTEGER), CAST('3.9' AS INTEGER), CAST(42 AS TEXT)
   (0, 12, 12, 3, '42')

-- CAST 결과의 typeof
   SELECT typeof(CAST('42' AS INTEGER)), typeof(CAST(42 AS TEXT)), typeof(CAST('42' AS NUMERIC)), typeof(CAST('4.5' AS NUMERIC))
   ('integer', 'text', 'integer', 'real')

================================================================
4) 비교에서의 친화성 — 컬럼과 리터럴을 비교할 때
================================================================

-- as_int(INTEGER 친화성) = '42' : 리터럴이 숫자로 바뀌어 비교된다
   SELECT id FROM sample WHERE as_int = '42' ORDER BY id
   (1,)
   (2,)
   (7,)

-- as_none(BLOB 친화성) = '42' : 변환이 없어 TEXT 와 INTEGER 는 다른 값이다
   SELECT id, typeof(as_none) FROM sample WHERE as_none = '42' ORDER BY id
   (1, 'text')

-- 친화성이 없는 표현식끼리: 42 = '42'
   SELECT 42 = '42', CAST(42 AS TEXT) = '42'
   (0, 1)

-- as_text(TEXT 친화성) = 42 : 숫자 쪽이 TEXT 로 바뀌어 비교된다
   SELECT id, as_text, typeof(as_text) FROM sample WHERE as_text = 42 ORDER BY id
   (1, '42', 'text')
   (7, '42', 'text')
