================================================================
1. 제약을 켜기만 하면 기존 행이 검사되는가
================================================================

-- 강제 설정 상태
   PRAGMA foreign_keys
   (1,)

-- 깨진 행이 그대로 남아 있다
   SELECT emp_id, emp_name, team_id FROM employee WHERE team_id = 30
   (3, '이서아', 30)

================================================================
2. PRAGMA foreign_key_check 가 돌려주는 네 칸
================================================================
   컬럼: ['table', 'rowid', 'parent', 'fkid']
    ('badge', None, 'employee', 0)
    ('employee', 3, 'team', 1)
    ('employee', 4, 'employee', 0)

-- 표 하나만 검사한다
   PRAGMA foreign_key_check(employee)
   ('employee', 3, 'team', 1)
   ('employee', 4, 'employee', 0)

================================================================
3. 네 번째 칸(fkid)은 foreign_key_list 의 id 와 같다
================================================================
   컬럼: ['id', 'seq', 'table', 'from', 'to', 'on_update', 'on_delete', 'match']
    (0, 0, 'employee', 'mentor_id', 'emp_id', 'NO ACTION', 'NO ACTION', 'NONE')
    (1, 0, 'team', 'team_id', 'team_id', 'NO ACTION', 'NO ACTION', 'NONE')

================================================================
4. rowid 로 깨진 행을 집어 낸다
================================================================

-- fkid 1 번(team_id) 을 위반한 행
   SELECT e.emp_id, e.emp_name, e.team_id FROM pragma_foreign_key_check('employee') AS c JOIN employee AS e ON e.rowid = c.rowid WHERE c.fkid = 1
   (3, '이서아', 30)

-- fkid 0 번(mentor_id) 을 위반한 행
   SELECT e.emp_id, e.emp_name, e.mentor_id FROM pragma_foreign_key_check('employee') AS c JOIN employee AS e ON e.rowid = c.rowid WHERE c.fkid = 0
   (4, '최민준', 99)

================================================================
5. WITHOUT ROWID 표는 두 번째 칸이 NULL 이다
================================================================
    ('badge', None, 'employee', 0)

================================================================
6. 고치고 다시 센다
================================================================

-- 남은 위반
   PRAGMA foreign_key_check
   (0행)

================================================================
7. 이제 새 위반은 들어오지 못한다
================================================================
   sqlite3.IntegrityError: FOREIGN KEY constraint failed

================================================================
8. 트랜잭션 안에서는 강제를 끄지 못한다
================================================================
   conn.in_transaction = True

-- OFF 를 넣었는데 값이 그대로다
   PRAGMA foreign_keys
   (1,)
   sqlite3.IntegrityError: FOREIGN KEY constraint failed

================================================================
9. 커밋해서 트랜잭션을 닫은 뒤에야 꺼진다
================================================================
   conn.in_transaction = False

-- 이번에는 꺼졌다
   PRAGMA foreign_keys
   (0,)

-- 그리고 깨진 행이 다시 쌓인다
   PRAGMA foreign_key_check(employee)
   ('employee', 7, 'team', 1)
