== 1) 묶음 비교는 왼쪽 컬럼부터 — NULL 이 끼면 ==
  sql> SELECT (2025, 12) < (2026, 1) AS a, (2026, 3) < (2026, 10) AS b, (2026, 3) = (2026, 3) AS c
  a | b | c
  1 | 1 | 1

  sql> SELECT (2026, NULL) = (2026, 3) AS eq_null, (2025, NULL) = (2026, 3) AS eq_false
  eq_null | eq_false
  NULL | 0

  sql> SELECT (2025, NULL) < (2026, 3) AS lt_decided, (2026, NULL) < (2026, 3) AS lt_unknown
  lt_decided | lt_unknown
  1 | NULL

  sql> SELECT (2026, NULL) IS (2026, NULL) AS is_same
  is_same
  1
