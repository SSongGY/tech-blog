== 1. 목록에 NULL 이 있을 때 IN / NOT IN 이 돌려주는 값 ==

[1-A 목록에 있는 값]
  SELECT 1 IN (1, 2, NULL), 1 NOT IN (1, 2, NULL)
  (1, 0)
  -> 1행

[1-B 목록에 없는 값]
  SELECT 3 IN (1, 2, NULL), 3 NOT IN (1, 2, NULL)
  (None, None)
  -> 1행

[1-C NULL 이 없는 목록]
  SELECT 3 IN (1, 2), 3 NOT IN (1, 2)
  (0, 1)
  -> 1행

[1-D 왼쪽이 NULL]
  SELECT NULL IN (1, 2), NULL NOT IN (1, 2)
  (None, None)
  -> 1행

[1-E 풀어 쓴 NOT IN]
  SELECT 3 <> 1 AND 3 <> 2 AND 3 <> NULL
  (None,)
  -> 1행

== 2. 산 적이 있는 고객 — IN, EXISTS, JOIN ==

[2-A IN]
  SELECT c.id, c.name FROM customer AS c WHERE c.id IN (SELECT p.customer_id FROM purchase AS p) ORDER BY c.id
  (1, '한지우')
  (2, '오민재')
  (3, '윤서아')
  -> 3행

[2-B EXISTS]
  SELECT c.id, c.name FROM customer AS c WHERE EXISTS (SELECT 1 FROM purchase AS p WHERE p.customer_id = c.id) ORDER BY c.id
  (1, '한지우')
  (2, '오민재')
  (3, '윤서아')
  -> 3행

[2-C INNER JOIN]
  SELECT c.id, c.name FROM customer AS c INNER JOIN purchase AS p ON p.customer_id = c.id ORDER BY c.id
  (1, '한지우')
  (1, '한지우')
  (2, '오민재')
  (3, '윤서아')
  (3, '윤서아')
  -> 5행

== 3. 한 번도 안 산 고객 — NOT IN, NOT EXISTS, LEFT JOIN ==

[3-A NOT IN]
  SELECT c.id, c.name FROM customer AS c WHERE c.id NOT IN (SELECT p.customer_id FROM purchase AS p) ORDER BY c.id
  -> 0행

[3-B NOT EXISTS]
  SELECT c.id, c.name FROM customer AS c WHERE NOT EXISTS (SELECT 1 FROM purchase AS p WHERE p.customer_id = c.id) ORDER BY c.id
  (4, '장하준')
  (5, '임채원')
  -> 2행

[3-C LEFT JOIN ... IS NULL]
  SELECT c.id, c.name FROM customer AS c LEFT JOIN purchase AS p ON p.customer_id = c.id WHERE p.id IS NULL ORDER BY c.id
  (4, '장하준')
  (5, '임채원')
  -> 2행

[3-D NOT IN + IS NOT NULL]
  SELECT c.id, c.name FROM customer AS c WHERE c.id NOT IN (SELECT p.customer_id FROM purchase AS p WHERE p.customer_id IS NOT NULL) ORDER BY c.id
  (4, '장하준')
  (5, '임채원')
  -> 2행
