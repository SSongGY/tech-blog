스키마와 데이터 — 아래 질의는 전부 이 표를 대상으로 한다
================================================================
[product]  4행
  컬럼         | 타입    | 제약
  -------------+---------+---------
  product_id   | INTEGER | PK
  product_name | TEXT    | NOT NULL
  unit_price   | INTEGER |

  product_id | product_name | unit_price
  -----------+--------------+-----------
           1 | 키보드       |      38000
           2 | 마우스       |      19000
           3 | 모니터       |     210000
           4 | 웹캠         |      55000

[supplier_price]  4행
  컬럼          | 타입    | 제약
  --------------+---------+---------
  supplier_name | TEXT    | NOT NULL
  product_id    | INTEGER | NOT NULL
  unit_price    | INTEGER | NOT NULL

  supplier_name | product_id | unit_price
  --------------+------------+-----------
  가나상사      |          1 |      36500
  가나상사      |          2 |      17800
  가나상사      |          3 |     198000
  다라유통      |          3 |     203000

================================================================
① 상관 서브쿼리 — WHERE 없이
================================================================
UPDATE product
   SET unit_price = (SELECT sp.unit_price
                       FROM supplier_price sp
                      WHERE sp.product_id = product.product_id
                        AND sp.supplier_name = '가나상사');

  갱신된 행 수: 4

-- 갱신 후 product
   (1, '키보드', 36500)
   (2, '마우스', 17800)
   (3, '모니터', 198000)
   (4, '웹캠', 'NULL')

================================================================
①-B 같은 문장을 unit_price NOT NULL 인 표에 돌리면
================================================================
  sqlite3.IntegrityError: NOT NULL constraint failed: product_strict.unit_price

-- 갱신 후 product_strict — 문장 전체가 되돌려졌다
   (1, '키보드', 38000)
   (2, '마우스', 19000)
   (3, '모니터', 210000)
   (4, '웹캠', 55000)

================================================================
② 상관 서브쿼리 — WHERE EXISTS 로 막기
================================================================
UPDATE product
   SET unit_price = (SELECT sp.unit_price
                       FROM supplier_price sp
                      WHERE sp.product_id = product.product_id
                        AND sp.supplier_name = '가나상사')
 WHERE EXISTS (SELECT 1
                 FROM supplier_price sp
                WHERE sp.product_id = product.product_id
                  AND sp.supplier_name = '가나상사');

  갱신된 행 수: 3

-- 갱신 후 product
   (1, '키보드', 36500)
   (2, '마우스', 17800)
   (3, '모니터', 198000)
   (4, '웹캠', 55000)

================================================================
③ UPDATE … FROM — 조인으로 갱신
================================================================
UPDATE product
   SET unit_price = sp.unit_price
  FROM supplier_price sp
 WHERE sp.product_id = product.product_id
   AND sp.supplier_name = '가나상사';

  갱신된 행 수: 3

-- 갱신 후 product

================================================================
⑤ FROM 절에 대상 표를 다시 적으면
================================================================
UPDATE product
   SET unit_price = sp.unit_price
  FROM product p
       JOIN supplier_price sp ON sp.product_id = p.product_id
 WHERE p.product_id = 1;

  갱신된 행 수: 4

-- 갱신 후 product
   (1, '키보드', 36500)
   (2, '마우스', 36500)
