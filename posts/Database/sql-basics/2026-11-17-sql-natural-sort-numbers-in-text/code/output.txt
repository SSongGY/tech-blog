[1] 파일 이름을 그대로 정렬한다
  sql> SELECT file_name FROM scan_file ORDER BY file_name
  결과: doc12, doc3, img007, img1, img10, img100, img2, img20, img7, img9

[2] 문자열 비교와 숫자 비교
  sql> SELECT '10' < '9', 10 < 9, typeof('10'), typeof(10)
    (1, 0, 'text', 'integer')

[3] 숫자만 든 문자열: 그대로 vs CAST
  sql> SELECT ticket_no FROM ticket ORDER BY ticket_no
  결과: 007, 10, 100, 12A, 25, 9
  sql> SELECT ticket_no FROM ticket ORDER BY CAST(ticket_no AS INTEGER)
  결과: 007, 9, 10, 12A, 25, 100

[4] CAST 가 돌려주는 값
  sql> SELECT CAST('12A' AS INTEGER), CAST('A12' AS INTEGER), CAST('007' AS INTEGER), CAST('1.10.2' AS INTEGER)
    (12, 0, 7, 1)

[5] 길이 먼저 정렬
  sql> SELECT ticket_no FROM ticket ORDER BY length(ticket_no), ticket_no
  결과: 9, 10, 25, 007, 100, 12A

[6] 접두어와 숫자를 떼어 정렬한다
  sql> SELECT file_name, rtrim(file_name, '0123456789') AS prefix, CAST(substr(file_name, length(rtrim(file_name, '0123456789')) + 1) AS INTEGER) AS num FROM scan_file WHERE file_id IN (3, 7, 9, 10)
    ('img10', 'img', 10)
    ('doc3', 'doc', 3)
    ('img007', 'img', 7)
    ('img7', 'img', 7)
  sql> SELECT file_name FROM scan_file ORDER BY rtrim(file_name, '0123456789'), CAST(substr(file_name, length(rtrim(file_name, '0123456789')) + 1) AS INTEGER)
  결과: doc3, doc12, img1, img2, img007, img7, img9, img10, img20, img100
  sql> SELECT file_name FROM scan_file ORDER BY rtrim(file_name, '0123456789'), CAST(substr(file_name, length(rtrim(file_name, '0123456789')) + 1) AS INTEGER), file_name
  결과: doc3, doc12, img1, img2, img007, img7, img9, img10, img20, img100

[7] 버전 문자열을 세 토막으로 나눈다
  sql> SELECT version FROM app_release ORDER BY version
  결과: 1.10.0, 1.10.12, 1.10.2, 1.2.0, 1.9.3, 2.0.0
  sql> WITH part AS ( SELECT version, substr(version, instr(version, '.') + 1) AS rest FROM app_release ) SELECT version, CAST(version AS INTEGER) AS major, CAST(rest AS INTEGER) AS minor, CAST(substr(rest, instr(rest, '.') + 1) AS INTEGER) AS patch FROM part ORDER BY major, minor, patch
    ('1.2.0', 1, 2, 0)
    ('1.9.3', 1, 9, 3)
    ('1.10.0', 1, 10, 0)
    ('1.10.2', 1, 10, 2)
    ('1.10.12', 1, 10, 12)
    ('2.0.0', 2, 0, 0)
