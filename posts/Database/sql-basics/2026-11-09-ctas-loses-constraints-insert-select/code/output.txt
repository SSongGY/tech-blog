  이름             | 대상 표 | 정의
  -----------------+---------+------------------------------------------------
  ix_product_price | product | CREATE INDEX ix_product_price ON product(price)

== 1) 정의 비교 — sqlite_schema ==
  CREATE TABLE product (
              product_id  INTEGER PRIMARY KEY,
              name        TEXT    NOT NULL COLLATE NOCASE,
              price       INTEGER NOT NULL DEFAULT 0 CHECK (price >= 0),
              sku         TEXT    UNIQUE
          )

  CREATE TABLE product_copy(
    product_id INT,
    name TEXT,
    price INT,
    sku TEXT
  )

== 2) 컬럼 정보와 인덱스 ==
  sql> SELECT name, type, "notnull", dflt_value, pk FROM pragma_table_info('product')
  name | type | notnull | dflt_value | pk
  'product_id' | 'INTEGER' | 0 | NULL | 1
  'name' | 'TEXT' | 1 | NULL | 0
  'price' | 'INTEGER' | 1 | '0' | 0
  'sku' | 'TEXT' | 0 | NULL | 0

  sql> SELECT name, "unique", origin FROM pragma_index_list('product')
  name | unique | origin

  'sqlite_autoindex_product_1' | 1 | 'u'

  sql> SELECT name, type, "notnull", dflt_value, pk FROM pragma_table_info('product_copy')
  name | type | notnull | dflt_value | pk
  'product_id' | 'INT' | 0 | NULL | 0
  'name' | 'TEXT' | 0 | NULL | 0
  'price' | 'INT' | 0 | NULL | 0
  'sku' | 'TEXT' | 0 | NULL | 0

  sql> SELECT name, "unique", origin FROM pragma_index_list('product_copy')
  name | unique | origin

  sql> SELECT count(*) AS copied_rows FROM product_copy

== 3) 원본이 막던 값을 넣어 본다 ==
  -- 대상: product
  sql> INSERT INTO product(product_id, name, price, sku) VALUES (1, 'dup', 10, 'D-01')
  IntegrityError: UNIQUE constraint failed: product.product_id

  sql> INSERT INTO product(product_id, name, price, sku) VALUES (10, NULL, 10, 'N-01')
  IntegrityError: NOT NULL constraint failed: product.name

  sql> INSERT INTO product(product_id, name, price, sku) VALUES (11, 'minus', -5, 'X-01')
  IntegrityError: CHECK constraint failed: price >= 0

  sql> INSERT INTO product(product_id, name, price, sku) VALUES (12, 'same sku', 10, 'A-01')
  IntegrityError: UNIQUE constraint failed: product.sku

  -- 대상: product_copy
  sql> INSERT INTO product_copy(product_id, name, price, sku) VALUES (1, 'dup', 10, 'D-01')
  (바뀐 행 수: 1)

  sql> INSERT INTO product_copy(product_id, name, price, sku) VALUES (10, NULL, 10, 'N-01')
  (바뀐 행 수: 1)

  sql> INSERT INTO product_copy(product_id, name, price, sku) VALUES (11, 'minus', -5, 'X-01')
  (바뀐 행 수: 1)

  sql> INSERT INTO product_copy(product_id, name, price, sku) VALUES (12, 'same sku', 10, 'A-01')
  (바뀐 행 수: 1)

  -- price 를 빼고 넣으면 기본값이 들어가는가
  sql> INSERT INTO product(name, sku) VALUES ('kiwi', 'K-01')
  (바뀐 행 수: 1)

  sql> INSERT INTO product_copy(name, sku) VALUES ('kiwi', 'K-01')
  (바뀐 행 수: 1)

  sql> SELECT product_id, name, price FROM product WHERE name = 'kiwi'
  product_id | name | price
  5 | 'kiwi' | 0

  sql> SELECT product_id, name, price FROM product_copy WHERE name = 'kiwi'
  product_id | name | price
  NULL | 'kiwi' | NULL

== 4) COLLATE NOCASE 가 사라지면 ==
