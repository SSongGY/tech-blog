  employee_id | name
  ------------+-------
  101         | 김하늘

[signup]  5행
  컬럼        | 타입    | 제약
  ------------+---------+---------
  employee_id | INTEGER | NOT NULL
  name        | TEXT    | NOT NULL
  email       | TEXT    |

  employee_id | name   | email
  ------------+--------+------------------
          101 | 김하늘 | sky@example.com

-- 1. 신청도 하고 출석도 한 사람 — INTERSECT
   SELECT employee_id, name FROM signup
   INTERSECT
   SELECT employee_id, name FROM attendance
   => employee_id | name
      101 | 김하늘
      103 | 박서준
   (2행)

-- 2. 신청만 하고 안 온 사람 — EXCEPT
   SELECT employee_id, name FROM signup
   EXCEPT
   SELECT employee_id, name FROM attendance
   => employee_id | name
      102 | 이도윤
      104 | 최지아
   (2행)

-- 3. 신청 없이 온 사람 — 순서를 뒤집은 EXCEPT
   SELECT employee_id, name FROM attendance
   EXCEPT
   SELECT employee_id, name FROM signup
   => employee_id | name
      105 | 정민재
   (1행)

-- 4. 중복 행 — UNION ALL 과 EXCEPT 의 행 수 비교
   SELECT 'signup 전체' AS source, COUNT(*) AS row_count FROM signup
   UNION ALL
   SELECT 'EXCEPT 결과', COUNT(*) FROM (
       SELECT employee_id, name FROM signup
       EXCEPT
       SELECT employee_id, name FROM attendance)
   => source | row_count
      signup 전체 | 5
      EXCEPT 결과 | 2
   (2행)

-- 5. INTERSECT ALL 은 있는가
   SELECT employee_id, name FROM signup
   INTERSECT ALL
   SELECT employee_id, name FROM attendance
   에러: near "ALL": syntax error

-- 6. NULL 이 든 행 — INTERSECT 는 NULL 끼리 같다고 본다
   SELECT employee_id, name, email FROM signup
   INTERSECT
   SELECT employee_id, name, email FROM attendance
   => employee_id | name | email
      101 | 김하늘 | sky@example.com
      103 | 박서준 | None
   (2행)

-- 7. 같은 일을 JOIN 으로 — email 까지 = 로 비교하면
   SELECT DISTINCT s.employee_id, s.name, s.email
   FROM signup AS s
   JOIN attendance AS a
     ON a.employee_id = s.employee_id
    AND a.name = s.name
    AND a.email = s.email
   => employee_id | name | email
      101 | 김하늘 | sky@example.com
   (1행)

-- 8. NOT IN 으로 차집합 — 하위 질의에 NULL 이 섞이면
   SELECT employee_id, name FROM signup
   WHERE email NOT IN (SELECT email FROM attendance)
   => employee_id | name
   (0행)

-- 9. 같은 질문을 EXCEPT 로
   SELECT email FROM signup
   EXCEPT
   SELECT email FROM attendance
   => email
      doyun@example.com
      jia@example.com
   (2행)

-- 10. 컬럼 수가 다르면
   SELECT employee_id, name FROM signup
   INTERSECT
   SELECT employee_id FROM attendance
   에러: SELECTs to the left and right of INTERSECT do not have the same number of result columns

-- 11. 컬럼 순서가 바뀌면 — 에러 없이 0행
   SELECT employee_id, name FROM signup
   INTERSECT
   SELECT name, employee_id FROM attendance
   => employee_id | name
   (0행)

-- 12. 타입이 다른 컬럼 — INTEGER 101 과 TEXT '101'
   SELECT employee_id, name FROM signup
   INTERSECT
   SELECT employee_id, name FROM attendance_sheet
   => employee_id | name
   (0행)

-- 13. 같은 비교를 JOIN = 로 하면
   SELECT DISTINCT s.employee_id, s.name
   FROM signup AS s
   JOIN attendance_sheet AS t
     ON t.employee_id = s.employee_id AND t.name = s.name
   => employee_id | name
      101 | 김하늘
   (1행)

-- 14. CAST 로 타입을 맞춘 INTERSECT
   SELECT employee_id, name FROM signup
   INTERSECT
   SELECT CAST(employee_id AS INTEGER), name FROM attendance_sheet
   => employee_id | name
      101 | 김하늘
   (1행)

-- 15. 섞어 쓰면 — signup EXCEPT attendance INTERSECT attendance_sheet
   SELECT employee_id FROM signup
   EXCEPT
   SELECT employee_id FROM attendance
   INTERSECT
   SELECT CAST(employee_id AS INTEGER) FROM attendance_sheet
   => employee_id
   (0행)

-- 16. 괄호로 오른쪽을 먼저 묶으려 하면
   SELECT employee_id FROM signup
   EXCEPT
   (SELECT employee_id FROM attendance
    INTERSECT
    SELECT CAST(employee_id AS INTEGER) FROM attendance_sheet)
   에러: near "(": syntax error

-- 17. 하위 질의로 오른쪽을 먼저 계산한다
   SELECT employee_id FROM signup
   EXCEPT
   SELECT employee_id FROM (
       SELECT employee_id FROM attendance
       INTERSECT
       SELECT CAST(employee_id AS INTEGER) FROM attendance_sheet)
   => employee_id
      102
      103
      104
   (3행)

-- 18. 개별 SELECT 에 ORDER BY 를 붙이면
   SELECT employee_id, name FROM signup ORDER BY name
   EXCEPT
   SELECT employee_id, name FROM attendance
   에러: ORDER BY clause should come after EXCEPT not before

-- 19. ORDER BY 는 맨 끝에 한 번 — 전체 결과를 정렬한다
   SELECT employee_id, name FROM signup
   EXCEPT
   SELECT employee_id, name FROM attendance
   ORDER BY name DESC
   => employee_id | name
      104 | 최지아
      102 | 이도윤
   (2행)
