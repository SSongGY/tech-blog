### 1. BETWEEN 의 경계값
-- 1. 금액 10000 이상 50000 이하 — BETWEEN
   SELECT order_id, customer_name, amount FROM order_log WHERE amount BETWEEN 10000 AND 50000 ORDER BY order_id
   => 2 | 이도윤 | 10000
   => 3 | 박서준 | 23500
   => 4 | 최지아 | 50000
   (3행)

-- 2. 같은 조건을 >= 와 <= 로 풀어 쓰면
   SELECT order_id, customer_name, amount FROM order_log WHERE amount >= 10000 AND amount <= 50000 ORDER BY order_id
   => 2 | 이도윤 | 10000
   => 3 | 박서준 | 23500
   => 4 | 최지아 | 50000
   (3행)

-- 3. 양끝을 뒤집으면 — BETWEEN 50000 AND 10000
   SELECT order_id, amount FROM order_log WHERE amount BETWEEN 50000 AND 10000
   => (0행)

-- 4. NOT BETWEEN — amount 가 NULL 인 행은 어느 쪽에도 없다
   SELECT order_id, customer_name, amount FROM order_log WHERE amount NOT BETWEEN 10000 AND 50000 ORDER BY order_id
   => 1 | 김하늘 | 9900
   => 5 | 정민재 | 50001
   (2행)

### 2. 날짜에 BETWEEN 을 쓸 때
-- 5. 3월 한 달 — BETWEEN '2026-03-01' AND '2026-03-31'
   SELECT order_id, customer_name, ordered_at FROM order_log WHERE ordered_at BETWEEN '2026-03-01' AND '2026-03-31' ORDER BY order_id
   => 1 | 김하늘 | 2026-03-01 09:12:30
   => 2 | 이도윤 | 2026-03-05 18:40:00
   => 3 | 박서준 | 2026-03-17 11:05:12
   (3행)

-- 6. 끝을 다음 달 1일로 바꾸고 < 로 끊으면
   SELECT order_id, customer_name, ordered_at FROM order_log WHERE ordered_at >= '2026-03-01' AND ordered_at < '2026-04-01' ORDER BY order_id
   => 1 | 김하늘 | 2026-03-01 09:12:30
   => 2 | 이도윤 | 2026-03-05 18:40:00
   => 3 | 박서준 | 2026-03-17 11:05:12
   => 4 | 최지아 | 2026-03-31 14:20:00
   => 5 | 정민재 | 2026-03-31 23:59:59
   (5행)

-- 7. 3월 31일 주문의 문자열 비교 결과
   SELECT '2026-03-31 14:20:00' <= '2026-03-31' AS le_cmp, '2026-03-31 14:20:00' < '2026-04-01' AS lt_cmp
   => 0 | 1
   (1행)

-- 8. 날짜 함수로 끊으면 — date(ordered_at)
   SELECT order_id, customer_name, ordered_at FROM order_log WHERE date(ordered_at) BETWEEN '2026-03-01' AND '2026-03-31' ORDER BY order_id
   => 1 | 김하늘 | 2026-03-01 09:12:30
   => 2 | 이도윤 | 2026-03-05 18:40:00
   => 3 | 박서준 | 2026-03-17 11:05:12
   => 4 | 최지아 | 2026-03-31 14:20:00
   => 5 | 정민재 | 2026-03-31 23:59:59
   (5행)

### 3. IN — 목록과 하위 질의
-- 9. 상태 목록 — IN
   SELECT order_id, customer_name, status FROM order_log WHERE status IN ('paid', 'shipped') ORDER BY order_id
   => 1 | 김하늘 | paid
   => 2 | 이도윤 | paid
   => 3 | 박서준 | shipped
   => 4 | 최지아 | paid
   (4행)

-- 10. 같은 조건을 OR 로 풀어 쓰면
   SELECT order_id, customer_name, status FROM order_log WHERE status = 'paid' OR status = 'shipped' ORDER BY order_id
   => 1 | 김하늘 | paid
   => 2 | 이도윤 | paid
   => 3 | 박서준 | shipped
   => 4 | 최지아 | paid
   (4행)

-- 11. 빈 목록 — IN ()
   SELECT order_id FROM order_log WHERE status IN ()
   => (0행)

-- 12. 환불 요청이 있는 주문 — IN 하위 질의
   SELECT order_id, customer_name FROM order_log WHERE order_id IN (SELECT order_id FROM refund_request) ORDER BY order_id
   => 3 | 박서준
   => 5 | 정민재
   (2행)

-- 13. 환불 요청이 없는 주문 — NOT IN 하위 질의 (NULL 이 섞여 있다)
   SELECT order_id, customer_name FROM order_log WHERE order_id NOT IN (SELECT order_id FROM refund_request) ORDER BY order_id
   => (0행)

-- 14. NULL 을 걸러낸 NOT IN
   SELECT order_id, customer_name FROM order_log WHERE order_id NOT IN ( SELECT order_id FROM refund_request WHERE order_id IS NOT NULL ) ORDER BY order_id
   => 1 | 김하늘
   => 2 | 이도윤
   => 4 | 최지아
   => 6 | 한소윤
   (4행)

-- 15. 같은 질문을 NOT EXISTS 로
   SELECT o.order_id, o.customer_name FROM order_log o WHERE NOT EXISTS ( SELECT 1 FROM refund_request r WHERE r.order_id = o.order_id ) ORDER BY o.order_id
   => 1 | 김하늘
   => 2 | 이도윤
   => 4 | 최지아
   => 6 | 한소윤
   (4행)

-- 16. IN 에 NULL 이 섞여도 찾는 값이 있으면 참이다
   SELECT 3 IN (1, 3, NULL) AS found, 2 IN (1, 3, NULL) AS not_found, 2 NOT IN (1, 3, NULL) AS negated
   => 1 | NULL | NULL
   (1행)

-- 18. bump() >= 1 AND bump() <= 5   → bump() 호출 2회

### 5. 인덱스를 어떻게 타는가
-- 19. BETWEEN — 범위
   SELECT order_id FROM order_log WHERE amount BETWEEN 10000 AND 50000
   SEARCH order_log USING COVERING INDEX ix_order_log_amount (amount>? AND amount<?)

-- 20. IN 목록 — 값 세 개
   SELECT order_id FROM order_log WHERE status IN ('paid','shipped','pending')
   SEARCH order_log USING COVERING INDEX ix_order_log_status (status=?)

-- 21. 컬럼을 함수로 감싸면
   SELECT order_id FROM order_log WHERE abs(amount) BETWEEN 10000 AND 50000
   SCAN order_log USING COVERING INDEX ix_order_log_amount
