-- 2. 같은 이메일이 두 번 이상인 것 — GROUP BY + HAVING
   SELECT email, COUNT(*) AS dup_count
   FROM member
   GROUP BY email
   HAVING COUNT(*) > 1
   => email | dup_count
      None | 2
      doyun@example.com | 2
      minjae@example.com | 3
   (3행)

-- 3. NULL 을 빼고 다시 센다
   SELECT email, COUNT(*) AS dup_count
   FROM member
   WHERE email IS NOT NULL
   GROUP BY email
   HAVING COUNT(*) > 1
   => email | dup_count
      doyun@example.com | 2
      minjae@example.com | 3
   (2행)

-- 4. 중복 묶음의 행을 전부 본다
   SELECT rowid, email, name, joined_at
   FROM member
   WHERE email IN (
       SELECT email FROM member
       WHERE email IS NOT NULL
       GROUP BY email HAVING COUNT(*) > 1)
   ORDER BY email, rowid
   => rowid | email | name | joined_at
      2 | doyun@example.com | 이도윤 | 2026-10-01 09:05
      3 | doyun@example.com | 이도윤 | 2026-10-01 09:05
      6 | minjae@example.com | 정민재 | 2026-10-03 14:00
      7 | minjae@example.com | 정민재 | 2026-10-03 14:01
      8 | minjae@example.com | 정민재 | 2026-10-05 08:30
   (5행)

-- 5. 묶음마다 남길 행 — MIN(rowid)
   SELECT email, MIN(rowid) AS keep_rowid, COUNT(*) AS dup_count
   FROM member
   GROUP BY email
   => email | keep_rowid | dup_count
      None | 5 | 2
      Jia@Example.com | 10 | 1
      doyun@example.com | 2 | 2
      jia@example.com | 4 | 1
      minjae@example.com | 6 | 3
      sky@example.com | 1 | 1
   (6행)

-- 6. 지울 행 미리 보기 — NULL 조건을 빠뜨린 판
   SELECT rowid, email, name
   FROM member
   WHERE rowid NOT IN (SELECT MIN(rowid) FROM member GROUP BY email)
   => rowid | email | name
      3 | doyun@example.com | 이도윤
      7 | minjae@example.com | 정민재
      8 | minjae@example.com | 정민재
      9 | None | 한유나
   (4행)

-- 7. 지울 행 미리 보기 — NULL 은 대상에서 뺀 판
   SELECT rowid, email, name
   FROM member
   WHERE email IS NOT NULL
     AND rowid NOT IN (SELECT MIN(rowid) FROM member GROUP BY email)
   => rowid | email | name
      3 | doyun@example.com | 이도윤
      7 | minjae@example.com | 정민재
      8 | minjae@example.com | 정민재
   (3행)

-- 8. 다른 기준으로 남기기 — 최근 가입 행을 1번으로
   SELECT rowid, email, joined_at,
          ROW_NUMBER() OVER (PARTITION BY email
                             ORDER BY joined_at DESC, rowid DESC) AS rn
   FROM member
   WHERE email IS NOT NULL
   ORDER BY email, rn
   => rowid | email | joined_at | rn
      10 | Jia@Example.com | 2026-10-06 12:00 | 1
      3 | doyun@example.com | 2026-10-01 09:05 | 1
      2 | doyun@example.com | 2026-10-01 09:05 | 2

   (8행)

-- 10. 트랜잭션 안에서 지운다 (7번 조건 그대로)

-- 11. 지운 뒤 다시 센다

   => email | dup_count
   (0행)

-- 15. 대소문자만 다른 이메일 — 아직 남아 있다
   SELECT rowid, email, name FROM member
   WHERE lower(email) = 'jia@example.com'
   => rowid | email | name
      4 | jia@example.com | 최지아
      10 | Jia@Example.com | 최지아
   (2행)

-- 16. lower() 로 묶으면 중복이 보인다
   SELECT lower(email) AS email_key, COUNT(*) AS dup_count
   FROM member
   WHERE email IS NOT NULL
   GROUP BY lower(email)
   HAVING COUNT(*) > 1
   => email_key | dup_count
      jia@example.com | 2
   (1행)

-- 17. lower(email) 기준으로 한 번 더 지운다
   DELETE FROM member
   WHERE email IS NOT NULL
     AND rowid NOT IN (SELECT MIN(rowid) FROM member GROUP BY lower(email))
   (바뀐 행 1)

-- 18. 표현식 UNIQUE 인덱스 — 대소문자 변형도 막는다
   CREATE UNIQUE INDEX ux_member_email_lower ON member (lower(email))
   (바뀐 행 -1)
