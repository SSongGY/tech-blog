== 1. 자르기 — SUBSTR, LENGTH, INSTR ==

[1-A 상품 코드 앞 3자리와 하이픈 뒤]
  SELECT product_code, SUBSTR(product_code, 1, 3) AS country, SUBSTR(product_code, 5) AS serial FROM member ORDER BY id
  ('KOR-1001', 'KOR', '1001')
  ('KOR-1002', 'KOR', '1002')
  ('USA-2001', 'USA', '2001')
  ('kor-1003', 'kor', '1003')
  ('FRA-3001', 'FRA', '3001')
  -> 5행

[1-B 음수 시작 위치 — 뒤에서부터 센다]
  SELECT product_code, SUBSTR(product_code, -4), SUBSTR(product_code, -4, 2) FROM member WHERE id <= 2
  ('KOR-1001', '1001', '10')
  ('KOR-1002', '1002', '10')
  -> 2행

[1-C 이메일에서 @ 앞뒤 나누기]
  SELECT email, INSTR(email, '@') AS at_pos, SUBSTR(email, 1, INSTR(email, '@') - 1) AS local_part, SUBSTR(email, INSTR(email, '@') + 1) AS domain FROM member ORDER BY id
  ('Doyun.Kim@Example.com', 10, 'Doyun.Kim', 'Example.com')
  ('seojun@example.com', 7, 'seojun', 'example.com')
  ('HAEUN@example.COM', 6, 'HAEUN', 'example.COM')
  ('yuna@example.com', 5, 'yuna', 'example.com')
  ('emile@example.fr', 6, 'emile', 'example.fr')
  -> 5행

[1-D LENGTH — 글자 수인가 바이트 수인가]
  SELECT member_name, LENGTH(member_name) AS chars, LENGTH(CAST(member_name AS BLOB)) AS utf8_bytes FROM member ORDER BY id
  ('김도윤', 3, 9)
  ('이서준', 3, 9)
  ('박하은', 3, 9)
  ('최유나', 3, 9)
  ('Émile', 5, 6)
  -> 5행

== 2. 붙이기 — ||, CONCAT, CONCAT_WS ==

[2-A || 로 이름과 메모 잇기]
  SELECT id, member_name || ' / ' || memo FROM member ORDER BY id
  (1, '김도윤 /   VIP  ')
  (2, '이서준 / \t신규')
  (3, None)
  (4, '최유나 / 재구매')
  (5, 'Émile / 해외')
  -> 5행

[2-B CONCAT 으로 같은 일]
  SELECT id, CONCAT(member_name, ' / ', memo) FROM member ORDER BY id
  (1, '김도윤 /   VIP  ')
  (2, '이서준 / \t신규')
  (3, '박하은 / ')
  (4, '최유나 / 재구매')
  (5, 'Émile / 해외')
  -> 5행

[2-C CONCAT_WS — 구분자를 한 번만 쓰고 NULL 은 건너뛴다]
  SELECT id, CONCAT_WS(' / ', member_name, memo, product_code) FROM member ORDER BY id
  (1, '김도윤 /   VIP   / KOR-1001')
  (2, '이서준 / \t신규 / KOR-1002')
  (3, '박하은 / USA-2001')
  (4, '최유나 / 재구매 / kor-1003')
  (5, 'Émile / 해외 / FRA-3001')
  -> 5행

== 3. 바꾸기 — REPLACE, TRIM, UPPER/LOWER ==

[3-A REPLACE — 대소문자를 구분한다]
  SELECT email, REPLACE(email, 'example.com', 'example.net') AS moved FROM member ORDER BY id
  ('Doyun.Kim@Example.com', 'Doyun.Kim@Example.com')
  ('seojun@example.com', 'seojun@example.net')
  ('HAEUN@example.COM', 'HAEUN@example.COM')
  ('yuna@example.com', 'yuna@example.net')
  ('emile@example.fr', 'emile@example.fr')
  -> 5행

[3-B TRIM — 기본은 공백만 지운다]
  SELECT id, '[' || memo || ']', '[' || TRIM(memo) || ']', '[' || TRIM(memo, ' ' || CHAR(9)) || ']' FROM member WHERE id IN (1, 2)
  (1, '[  VIP  ]', '[VIP]', '[VIP]')
  (2, '[\t신규]', '[\t신규]', '[신규]')
  -> 2행

[3-C UPPER/LOWER — ASCII 밖의 글자는 그대로]
  SELECT member_name, UPPER(member_name), LOWER(member_name), LOWER(email) FROM member ORDER BY id
  ('김도윤', '김도윤', '김도윤', 'doyun.kim@example.com')
  ('이서준', '이서준', '이서준', 'seojun@example.com')
  ('박하은', '박하은', '박하은', 'haeun@example.com')
  ('최유나', '최유나', '최유나', 'yuna@example.com')
  ('Émile', 'ÉMILE', 'Émile', 'emile@example.fr')
  -> 5행

== 4. WHERE 절의 함수와 인덱스 ==
  (JPN- 코드 회원 2000행을 더 넣고 ANALYZE 했다)

[4-A 컬럼에 SUBSTR 을 씌운 조건]  결과 2행 · VM 명령 8,035개

  QUERY PLAN

[4-B LIKE 'KOR%']  결과 3행 · VM 명령 8,035개

  QUERY PLAN
  `--SCAN member USING COVERING INDEX ix_member_product_code

[4-C 범위 조건으로 바꿔 쓴 접두 검색]  결과 2행 · VM 명령 23개
  SELECT id FROM member WHERE product_code >= 'KOR' AND product_code < 'KOS'
  QUERY PLAN
  `--SEARCH member USING COVERING INDEX ix_member_product_code (product_code>? AND product_code<?)

[4-D GLOB 'KOR*']  결과 2행 · VM 명령 24개
  SELECT id FROM member WHERE product_code GLOB 'KOR*'
  QUERY PLAN
  `--SEARCH member USING COVERING INDEX ix_member_product_code (product_code>? AND product_code<?)

[4-E 컬럼에 LOWER 를 씌운 이메일 검색]  결과 1행 · VM 명령 8,031개

[4-F 식 인덱스 LOWER(email) 을 만든 뒤 같은 질의]  결과 1행 · VM 명령 16개

[4-G 식 인덱스는 그대로 두고 UPPER 로 찾으면]  결과 1행 · VM 명령 8,031개
