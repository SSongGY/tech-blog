SQL> -- tbsql 스크립트 모드의 두 가지 제약 때문에 이 파일의 형식이 이렇다.
SQL> --   1) 끝에 EXIT가 없으면 프롬프트에서 대기한다. 무인 실행이 멈춘다.
SQL> --   2) 문장 뒤에 같은 줄로 `-- 주석`을 달면 다음 문장이 TBR-8004로 깨진다.
SQL> --      그래서 모든 주석을 독립된 줄에 둔다. code/README.md에 재현 예가 있다.

SQL> --    MAXVALUE 9999999999999999999999999999 (9가 28개. INT64_MAX가 아니다)

SQL> CREATE SEQUENCE order_seq;

SQL> SELECT order_seq.NEXTVAL FROM dual;

SQL> -- ================================================================
SQL> -- 4. 구성요소 표에 없는 제약들. 전부 생성 단계에서 막힌다.
SQL> -- ================================================================
SQL>
SQL> -- 오름차순 CYCLE에는 MAXVALUE가 필요하다 → TBR-7136
SQL> CREATE SEQUENCE bad_cycle_seq START WITH 1 MINVALUE 1 INCREMENT BY 1 CYCLE;
TBR-7136: In ascending sequences, the CYCLE option must be used with MAXVALUE.

SQL>
SQL> -- 캐시 크기는 한 사이클보다 작아야 한다 → TBR-7340 (사이클 12개에 캐시 20개)
SQL> CREATE SEQUENCE bad_cache_seq START WITH 1 MINVALUE 1 MAXVALUE 12
   2     INCREMENT BY 1 CYCLE CACHE 20;
TBR-7340: Sequence cache size must be less than one cycle.

SQL>
SQL> -- INCREMENT는 MAXVALUE - MINVALUE보다 작아야 한다 → TBR-7424
SQL> CREATE SEQUENCE big_step_seq MINVALUE 1 MAXVALUE 10 INCREMENT BY 100;
TBR-7424: INCREMENT must be less than MAXVALUE minus MINVALUE.
at line 1, column 53 of null:
CREATE SEQUENCE big_step_seq MINVALUE 1 MAXVALUE 10 INCREMENT BY 100
                                                    ^^^^^^^^^^^^^^^^

SQL>
SQL> -- RESTART는 CREATE에서 쓸 수 없다 → TBR-7615
SQL> CREATE SEQUENCE bad_restart_seq RESTART START WITH 5;
TBR-7615: Restart option is not allowed for create sequence

SQL>
SQL> -- 사이클보다 작은 캐시는 통과한다
SQL> CREATE SEQUENCE slot_seq START WITH 1 MINVALUE 1 MAXVALUE 12
   2     INCREMENT BY 1 CYCLE CACHE 4;

Sequence 'SLOT_SEQ' created.

SQL>

SQL> -- 6. 컬럼 DEFAULT에 NEXTVAL. 매뉴얼 두 페이지가 반대로 적는 항목인데

SQL> -- 8. ALTER.
SQL> --    START WITH는 못 바꾼다 → TBR-7008
SQL> --    RESTART는 START WITH로 준 1000이 아니라 MINVALUE 기본값 1로 돌아간다.
SQL> --    특정 값이 필요하면 RESTART START WITH로 직접 준다.
SQL> -- ================================================================
SQL> CREATE SEQUENCE invoice_seq START WITH 1000 INCREMENT BY 1 CACHE 100;
