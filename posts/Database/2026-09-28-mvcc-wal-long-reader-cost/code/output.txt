== 1. rollback journal(DELETE) 모드 — 읽기가 쓰기를 막는다 ==
  journal_mode = delete
  읽기   BEGIN                                            성공
  읽기   SELECT balance FROM account WHERE id = 1         성공 -> [(1000,)]
  쓰기   BEGIN                                            성공
  쓰기   UPDATE account SET balance = 900 WHERE id = 1    성공
  쓰기   COMMIT                                           에러: database is locked
  읽기   COMMIT                                           성공
  쓰기   COMMIT                                           성공
  읽기   SELECT balance FROM account WHERE id = 1         성공 -> [(900,)]

== 2. WAL 모드 — 쓰기가 통과하고 읽기는 시작 시점을 본다 ==
  journal_mode = wal
  읽기   BEGIN                                            성공
  읽기   SELECT balance FROM account WHERE id = 1         성공 -> [(1000,)]
  쓰기   BEGIN                                            성공
  쓰기   UPDATE account SET balance = 900 WHERE id = 1    성공
  쓰기   COMMIT                                           성공
  쓰기   SELECT balance FROM account WHERE id = 1         성공 -> [(900,)]
  읽기   SELECT balance FROM account WHERE id = 1         성공 -> [(1000,)]
  읽기   COMMIT                                           성공
  읽기   SELECT balance FROM account WHERE id = 1         성공 -> [(900,)]

== 3. 대가 — 열린 읽기 트랜잭션이 체크포인트를 멈춘다 ==
  page_size = 4096, wal_autocheckpoint = 1000
